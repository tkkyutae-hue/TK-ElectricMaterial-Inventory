import { branchFetch, documentBranchPrefix } from "./branchTransport";

const pending = new Map<string, string>();
const uncertain = new Set<string>();
const flights = new Map<string, Promise<any>>();
let mediaGeneration = 0;
const downloads = new Set<string>();
if (typeof window !== "undefined") window.addEventListener("inventory:branch-session-ended", () => {
  mediaGeneration++;
  pending.clear();
  uncertain.clear();
  flights.clear();
  for (const url of downloads) URL.revokeObjectURL(url);
  downloads.clear();
});
function contextGuard() {
  const generation = mediaGeneration;
  return () => {
    if (generation !== mediaGeneration) throw Object.assign(Error("BRANCH_CONTEXT_CHANGED"), { code: "BRANCH_CONTEXT_CHANGED" });
  };
}
export function branchMediaSource(url?: string | null): string | undefined {
  if (!url) return undefined;
  if (!documentBranchPrefix) return url;
  if (documentBranchPrefix === "/branches/indiana"
    && /^\/branches\/indiana\/api\/catalog-images\/(?:item|category|group)\/[1-9]\d*\/[0-3]$/.test(url)) return url;
  if (documentBranchPrefix === "/branches/savannah"
    && /^\/branches\/savannah\/api\/legacy-media\/items\/[1-9]\d*\/images\/[1-9]\d*\/(?:content|download|thumbnail)$/.test(url)) return url;
  return new RegExp("^" + documentBranchPrefix + "/api/media/[a-f0-9-]{36}/(?:content|thumbnail|download)$", "i").test(url)
    ? url : undefined;
}

export function mediaErrorState(error: unknown): "failed" | "forbidden" | "unconfigured" | "missing" | "unknown" {
  const code = String((error as any)?.code ?? (error as any)?.message ?? "");
  const status = (error as any)?.status;
  if (/NOT_CONFIGURED|REVIEW_REQUIRED|CATALOG.*(?:NOT_READY|UNAVAILABLE)/.test(code)) return "unconfigured";
  // Only an explicit post-authorization object miss is evidence of a lost file.
  // Generic 404 is deliberately indistinguishable from denied/not-visible records.
  if (code === "BRANCH_MEDIA_OBJECT_MISSING") return "missing";
  if (/RESULT_UNKNOWN/.test(code)) return "unknown";
  if (status === 401 || status === 403 || status === 404
    || /ACCESS_DENIED|ACCOUNT_|CONTEXT_CHANGED|REAUTHENTICATION|NOT_FOUND|URL_NOT_SCOPED|SOURCE_INVALID/.test(code)) return "forbidden";
  return "failed";
}
export function mediaCanRetry(error: unknown) {
  const code = String((error as any)?.code ?? (error as any)?.message ?? "");
  return mediaErrorState(error) === "failed"
    && !/SIZE_INVALID|TYPE_INVALID|TARGET_INVALID|LIMIT|GALLERY_FULL|CONFLICT|READ_ONLY|INVALID/.test(code);
}
export function mediaErrorMessage(error: unknown, lang: string) {
  const messages = mediaMessages[lang as keyof typeof mediaMessages] ?? mediaMessages.en;
  const code = String((error as any)?.code ?? (error as any)?.message ?? "");
  const state = mediaErrorState(error);
  if (state === "unconfigured") return messages.unavailable;
  if (state !== "failed") return messages[state];
  if (/SIZE_INVALID|TYPE_INVALID/.test(code)) return messages.invalid;
  if (/LIMIT|GALLERY_FULL|EXPORT_TOO_LARGE/.test(code)) return messages.limit;
  if (/READ_ONLY/.test(code)) return messages.readOnly;
  if (/CONFLICT|INVALID/.test(code)) return messages.conflict;
  return messages.failed;
}
export async function branchMediaResult(response: Response) {
  const body = await response.json().catch(() => null);
  if (!response.ok) throw Object.assign(Error(body?.code ?? "BRANCH_MEDIA_UNAVAILABLE"), {
    code: body?.code ?? "BRANCH_MEDIA_UNAVAILABLE", status: response.status,
  });
  if (body === null) throw Error("BRANCH_MEDIA_UNAVAILABLE");
  return body;
}
export async function branchGallery(itemId: number) {
  const check = contextGuard();
  const body = await branchMediaResult(await branchFetch(`/api/inventory/${itemId}/images`, { credentials: "include" }));
  check();
  if (!Array.isArray(body)) throw Error("BRANCH_MEDIA_UNAVAILABLE");
  return body;
}
export async function fetchBranchMediaBlob(url: string): Promise<Blob> {
  const check = contextGuard();
  if (!documentBranchPrefix || branchMediaSource(url) !== url) throw Error("BRANCH_MEDIA_SOURCE_INVALID");
  const response = await branchFetch(url, { credentials: "include", cache: "no-store" });
  if (!response.ok) await branchMediaResult(response);
  const blob = await response.blob();
  check();
  if (!blob.size || !/^image\/(?:jpeg|png|webp)$/.test(blob.type)) throw Error("BRANCH_MEDIA_UNAVAILABLE");
  return blob;
}
async function saveDownload(path: string, filename: string) {
  const check = contextGuard();
  const response = await branchFetch(path, { credentials: "include", cache: "no-store" });
  if (!response.ok) await branchMediaResult(response);
  const blob = await response.blob();
  check();
  if (!blob.size || !["image/jpeg", "image/png", "image/webp", "application/pdf", "application/zip"].includes(blob.type)) throw Error("BRANCH_MEDIA_UNAVAILABLE");
  const url = URL.createObjectURL(blob);
  downloads.add(url);
  const link = document.createElement("a");
  link.href = url; link.download = filename;
  document.body.appendChild(link); link.click(); link.remove();
  // Browser delivery is not proof of saving the file on the user's device.
  setTimeout(() => { URL.revokeObjectURL(url); downloads.delete(url); }, 30_000);
}
export async function downloadBranchMedia(url: string, filename = "attachment") {
  if (!documentBranchPrefix || branchMediaSource(url) !== url || !/\/(?:content|download|thumbnail)$/.test(url)) throw Error("BRANCH_MEDIA_SOURCE_INVALID");
  return saveDownload(url.replace(/\/(?:content|thumbnail)$/, "/download"), filename);
}
function validTarget(kind: string, targetId: number) {
  if (!["item", "category", "request", "project", "movement"].includes(kind) || !Number.isSafeInteger(targetId) || targetId <= 0) throw Error("BRANCH_MEDIA_TARGET_INVALID");
}
export async function downloadBranchArchive(kind: string, targetId: number) {
  if (!documentBranchPrefix) throw Error("BRANCH_MEDIA_CONTEXT_REQUIRED");
  validTarget(kind, targetId);
  return saveDownload(`/api/media/targets/${kind}/${targetId}/export`, "attachments.zip");
}
export async function uploadBranchFile(file: File, kind: string, targetId: number) {
  const check = contextGuard();
  if (!documentBranchPrefix) throw Error("BRANCH_MEDIA_CONTEXT_REQUIRED");
  validTarget(kind, targetId);
  if (!file.size || file.size > 5 * 1024 * 1024) throw Error("BRANCH_MEDIA_SIZE_INVALID");
  if (!["image/jpeg", "image/png", "image/webp", "application/pdf"].includes(file.type)) throw Error("BRANCH_MEDIA_TYPE_INVALID");
  const data = await file.arrayBuffer();
  check();
  const digest = Array.from(new Uint8Array(await crypto.subtle.digest("SHA-256", data)), b => b.toString(16).padStart(2, "0")).join("");
  check();
  const key = [kind, targetId, file.type, digest].join(":");
  if (flights.has(key)) return flights.get(key);
  const operationId = pending.get(key) ?? crypto.randomUUID();
  pending.set(key, operationId);
  const command = (async () => {
    try {
      const body = await branchMediaResult(await branchFetch(`/api/media/targets/${kind}/${targetId}`, {
        method: "POST", body: data, headers: { "Content-Type": file.type, "X-Media-Operation": operationId },
      }));
      check();
      if (body.state !== "ready" || !branchMediaSource(body.url)) throw Error("BRANCH_MEDIA_RESULT_UNKNOWN");
      pending.delete(key);
      uncertain.delete(key);
      return body as { id: string; url: string; state: string };
    } catch (e: any) {
      check();
      if (!e.status || e.status >= 500) {
        try {
          const confirmed = await branchMediaResult(await branchFetch(`/api/media/operations/${operationId}`));
          check();
          if (confirmed.state === "ready" && branchMediaSource(confirmed.url)) {
            pending.delete(key);
            uncertain.delete(key);
            return confirmed;
          }
        } catch { /* Absence may race an unfinished command; it does not prove failure. */ }
        check();
        uncertain.add(key);
        throw Object.assign(Error("BRANCH_MEDIA_RESULT_UNKNOWN"), { code: "BRANCH_MEDIA_RESULT_UNKNOWN", operationId });
      }
      // A later rejection does not prove an earlier lost reply was not applied.
      // Keep that original operation unresolved; do not unlock a new submission.
      if (uncertain.has(key)) throw Object.assign(Error("BRANCH_MEDIA_RESULT_UNKNOWN"), {
        code: "BRANCH_MEDIA_RESULT_UNKNOWN", operationId, canRetry: false,
      });
      throw e;
    }
  })();
  flights.set(key, command);
  try { return await command; }
  finally { if (flights.get(key) === command) flights.delete(key); }
}
