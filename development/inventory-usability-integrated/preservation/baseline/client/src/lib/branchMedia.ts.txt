import { branchFetch, documentBranchPrefix } from "./branchTransport";

const pending = new Map<string, string>();
if (typeof window !== "undefined") window.addEventListener("inventory:branch-session-ended", () => pending.clear());
export function branchMediaSource(url?: string | null): string | undefined {
  if (!url) return undefined;
  if (!documentBranchPrefix) return url;
  if (documentBranchPrefix === "/branches/indiana"
    && /^\/branches\/indiana\/api\/catalog-images\/(?:item|category|group)\/[1-9]\d*\/[0-3]$/.test(url)) return url;
  if (documentBranchPrefix === "/branches/savannah"
    && /^\/branches\/savannah\/api\/legacy-media\/items\/[1-9]\d*\/images\/[1-9]\d*\/(?:content|download|thumbnail)$/.test(url)) return url;
  return new RegExp("^" + documentBranchPrefix + "/api/media/[a-f0-9-]{36}/(?:content|thumbnail|download)$", "i").test(url)
    ? url : undefined;
}

export function mediaErrorMessage(error: unknown, lang: string) {
  const messages = mediaMessages[lang as keyof typeof mediaMessages] ?? mediaMessages.en;
  const code = (error as any)?.code ?? (error as Error)?.message ?? "";
  if (code.includes("NOT_CONFIGURED")) return messages.unavailable;
  if (code.includes("NOT_FOUND") || code.includes("OBJECT_MISSING")) return messages.missing;
  if (code.includes("RESULT_UNKNOWN")) return messages.unknown;
  return messages.failed;
}
async function result(response: Response) {
  const body = await response.json();
  if (!response.ok) throw Object.assign(Error(body?.code ?? "BRANCH_MEDIA_UNAVAILABLE"), { code: body?.code, status: response.status });
  return body;
}
export async function branchGallery(itemId: number) {
  return result(await branchFetch(`/api/inventory/${itemId}/images`, { credentials: "include" }));
}
export async function uploadBranchFile(file: File, kind: string, targetId: number) {
  if (!documentBranchPrefix) throw Error("BRANCH_MEDIA_CONTEXT_REQUIRED");
  if (!["item", "category", "request", "project", "movement"].includes(kind) || !Number.isSafeInteger(targetId) || targetId <= 0) throw Error("BRANCH_MEDIA_TARGET_INVALID");
  if (!file.size || file.size > 5 * 1024 * 1024) throw Error("BRANCH_MEDIA_SIZE_INVALID");
  const data = await file.arrayBuffer();
  const digest = Array.from(new Uint8Array(await crypto.subtle.digest("SHA-256", data)), b => b.toString(16).padStart(2, "0")).join("");
  const key = [kind, targetId, file.type, digest].join(":");
  const operationId = pending.get(key) ?? crypto.randomUUID();
  pending.set(key, operationId);
  try {
    const body = await result(await branchFetch(`/api/media/targets/${kind}/${targetId}`, {
      method: "POST", body: data, headers: { "Content-Type": file.type, "X-Media-Operation": operationId },
    }));
    if (body.state !== "ready" || !branchMediaSource(body.url)) throw Error("BRANCH_MEDIA_RESULT_UNKNOWN");
    pending.delete(key);
    return body as { id: string; url: string; state: string };
  } catch (e: any) {
    // Confirm after a lost response, never infer failure from network/JSON errors.
    if (!e.status || e.status >= 500) {
      try {
        const confirmed = await result(await branchFetch(`/api/media/operations/${operationId}`));
        if (confirmed.state === "ready" && branchMediaSource(confirmed.url)) {
          pending.delete(key);
          return confirmed;
        }
      } catch { /* absence can race an unfinished command; do not invent failure */ }
      throw Object.assign(Error("BRANCH_MEDIA_RESULT_UNKNOWN"), { code: "BRANCH_MEDIA_RESULT_UNKNOWN", operationId });
    }
    throw e; // Keep operation ID for an explicit same-file retry.
  }
}
