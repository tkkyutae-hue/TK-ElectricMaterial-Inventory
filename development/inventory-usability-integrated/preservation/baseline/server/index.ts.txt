import express, { type Request, Response, NextFunction } from "express";
import { initializeApplication } from "./bootstrap";
import { readStartupPlan } from "./branches/startupPlan";
import { runtimeEnvironmentReader } from "./branches/developmentRuntimeEnvironment";
import { loadDevelopmentSharedMedia } from "./branches/sharedMediaStorage";
import { loadDevelopmentCatalogAssets } from "./branches/catalogPrivateAssets";
import { loadDevelopmentCatalogDeployment } from "./branches/catalogDeployment";
import { serveStatic } from "./static";
import { createServer } from "http";

const app = express();
const httpServer = createServer(app);

declare module "http" {
  interface IncomingMessage {
    rawBody: unknown;
  }
}

export function log(message: string, source = "express") {
  const formattedTime = new Date().toLocaleTimeString("en-US", {
    hour: "numeric",
    minute: "2-digit",
    second: "2-digit",
    hour12: true,
  });
  console.log(`${formattedTime} [${source}] ${message}`);
}

// ─── Boot-time environment validation ────────────────────────────────────────
// Only the explicit mode switch is read here; shared mode is fully validated
// (restricted connections, approval, master designation) before app setup.
const startupMode = process.env.TK_SERVER_MODE === undefined || process.env.TK_SERVER_MODE === "legacy" ? "legacy" : "shared";
const REQUIRED_ENV = ["DATABASE_URL", "SESSION_SECRET"] as const;
// Shared mode validates only its explicit restricted connections. It must
// neither require nor read the legacy owner's DATABASE_URL credentials.
const missing = startupMode === "legacy" ? REQUIRED_ENV.filter((k) => !process.env[k]) : [];
if (missing.length > 0) {
  console.error(`[boot] FATAL — missing required env vars: ${missing.join(", ")}`);
  process.exit(1);
}

app.use(
  express.json({
    limit: "50mb",
    verify: (req, _res, buf) => {
      req.rawBody = buf;
    },
  }),
);

app.use(express.urlencoded({ extended: false }));

app.use((req, res, next) => {
  const start = Date.now();
  const path = req.path;

  res.on("finish", () => {
    const duration = Date.now() - start;
    if (path.startsWith("/api")) {
      log(`${req.method} ${path} ${res.statusCode} in ${duration}ms`);
    }
  });

  next();
});

(async () => {
  const startupPlan = readStartupPlan(await runtimeEnvironmentReader(process.env));
  const sharedMedia = await loadDevelopmentSharedMedia(startupPlan, process.env);
  const catalogAssets = await loadDevelopmentCatalogAssets(startupPlan, process.env);
  const catalogDeployment = await loadDevelopmentCatalogDeployment(startupPlan, process.env, catalogAssets.catalogImages);
  await initializeApplication(app, httpServer, startupPlan, { log, ...sharedMedia, ...catalogAssets, ...catalogDeployment });

  app.use((err: any, _req: Request, res: Response, next: NextFunction) => {
    const status = err.status || err.statusCode || 500;
    const message = err.message || "Internal Server Error";
    console.error("Internal Server Error:", err);
    if (res.headersSent) return next(err);
    return res.status(status).json({ message });
  });

  if (process.env.NODE_ENV === "production") {
    serveStatic(app);
  } else {
    const { setupVite } = await import("./vite");
    await setupVite(httpServer, app);
  }

  const port = parseInt(process.env.PORT || "5000", 10);
  httpServer.listen(
    { port, host: "0.0.0.0", reusePort: true },
    () => { log(`serving on port ${port}`); },
  );
})();
