export function CatalogPreparationPanel() {
  const queries = useQueryClient();
  const { isAdminRole, user } = useAuth(), { lang } = useLanguage();
  const t = labels[lang] ?? labels.en;
  const it = importLabels[lang] ?? importLabels.en;
  const st = sourceLabels[lang] ?? sourceLabels.en;
  const [sourceBusy, setSourceBusy] = useState(false);
  const [sourceIssues, setSourceIssues] = useState<string[] | null>(null);
  const sourceRequest = useRef(0);
  const account = useRef(user?.id);
  const [, setAccountVersion] = useState(0);
  const [input, setInput] = useState(""), [busy, setBusy] = useState(false);
  const [result, setResult] = useState<number | null>(null), [error, setError] = useState<string | null>(null);
  const epoch = useRef(0);
  const [ready, setReady] = useState(false), [confirmed,setConfirmed] = useState(false);
  const [outcome,setOutcome] = useState<string | null>(null), [pending,setPending] = useState(false);
  const operation = useRef<{ id:string; input:string; uncertain:boolean } | null>(null);
  function endPanelSession() {
    epoch.current++;setInput("");setResult(null);setError(null);setReady(false);
    setConfirmed(false);setOutcome(null);operation.current=null;setPending(false);
    sourceRequest.current++;setSourceBusy(false);setSourceIssues(null);
    setAccountVersion(version => version + 1);
  }
  useEffect(() => {
    window.addEventListener("inventory:branch-session-ended", endPanelSession);
    return () => { epoch.current++;sourceRequest.current++;window.removeEventListener("inventory:branch-session-ended", endPanelSession); };
  }, []);
  useEffect(() => {
    if (account.current !== user?.id) {
      account.current = user?.id;
      endPanelSession();
    }
  }, [user?.id]);
  if (account.current !== user?.id || !isAdminRole || documentBranchPrefix !== "/branches/indiana") return null;
  async function loadSource() {
    if (sourceBusy) return;
    if (operation.current) { setError(it.locked); return; }
    if (input.trim() && !window.confirm(st.replace)) return;
    const request = ++sourceRequest.current, token = epoch.current;
    setSourceBusy(true);setError(null);setReady(false);setConfirmed(false);
    try {
      const response = await branchFetch("/api/admin/catalog-preparation/source-review", {
        method: "POST", headers: { "Content-Type": "application/json" },
        // Source/DB identity, actor and approved assets are never browser inputs.
        body: JSON.stringify({ locations: [], suppliers: [], images: [] }),
      });
      const data = await response.json();
      if (request !== sourceRequest.current) return;
      if (token !== epoch.current) { setError(st.changed); return; }
      if (response.status === 401 || response.status === 403) { setError(st.denied); return; }
      if (!response.ok && data.code === "CATALOG_SOURCE_NOT_CONFIGURED") { setError(st.unavailable); return; }
      if (!response.ok || data.kind !== "indiana-source-catalog-review" || data.executable !== false
        || !data.input || !Array.isArray(data.input.items) || !Array.isArray(data.issues)
        || data.issues.some((issue: any) => typeof issue?.code !== "string")) throw Error();
      const nextInput = JSON.stringify(data.input, null, 2);
      epoch.current++;setInput(nextInput);setResult(null);setReady(false);setConfirmed(false);
      setSourceIssues(data.issues.map((issue: { code: string }) => issue.code));
    } catch {
      if (request === sourceRequest.current && token === epoch.current) setError(st.failed);
    } finally { if (request === sourceRequest.current) setSourceBusy(false); }
  }
  async function preview() {
    const token = epoch.current;
    setBusy(true); setError(null); setResult(null);setReady(false);setConfirmed(false);
    try {
      const parsed = JSON.parse(input);
      const response = await branchFetch("/api/admin/catalog-preparation/preview", {
        method: "POST", headers: { "Content-Type": "application/json" },
        body: JSON.stringify(parsed),
      });
      const data = await response.json();
      if (token !== epoch.current) { setError(t.changed); return; }
      if (!response.ok && typeof data.code === "string" && /CATALOG_.*(?:MAPPING|REFERENCE)/.test(data.code)) {
        setError(it.mapping);return;
      }
      if (response.status === 403) {setError(it.denied);return;}
      if (!response.ok || data.kind !== "new-indiana-catalog-preview" || data.executable !== false || !Array.isArray(data.items)) throw Error();
      setResult(data.items.length);
      setReady(data.importReady === true);setConfirmed(false);
    } catch { if (token === epoch.current) setError(t.failed); }
    finally { setBusy(false); }
  }
  async function importCatalog(check = false) {
    if (pending || sourceBusy) return;
    if (!operation.current) {
      if (check || !ready || !confirmed) return;
      operation.current = {id:crypto.randomUUID(),input,uncertain:false};
    }
    const op = operation.current, token = epoch.current;
    setPending(true);setOutcome(null);
    try {
      const response = await branchFetch(check ? `/api/admin/catalog-preparation/operations/${op.id}` : "/api/admin/catalog-preparation/import",
        check ? undefined : {method:"POST",headers:{"Content-Type":"application/json"},
          body:JSON.stringify({operationId:op.id,catalog:JSON.parse(op.input)})});
      const body = await response.json();
      // Account/document invalidation must not reintroduce old outcome/input.
      if (operation.current !== op) return;
      if (!response.ok) {
        if (check && response.status === 403) {op.uncertain=true;setOutcome(it.denied);return;}
        if (response.status >= 500 || op.uncertain || check) throw Error();
        operation.current=null;setOutcome(it.failure);return;
      }
      if (body.state === "applied" && body.operationId === op.id) {
        operation.current=null;setOutcome(it.applied);
        // Refresh affected views, not drafts. A refetch failure cannot turn a
        // confirmed committed command into an unknown/failed save.
        void queries.invalidateQueries({queryKey:["/api/inventory"]}).catch(()=>{});
        void queries.invalidateQueries({queryKey:[api.items.list.path]}).catch(()=>{});
        void queries.invalidateQueries({queryKey:[api.categories.list.path]}).catch(()=>{});
        void queries.invalidateQueries({queryKey:["/api/inventory/categories/summary"]}).catch(()=>{});
        if (token === epoch.current) {setReady(false);setConfirmed(false);}
      } else if (body.state === "not-recorded") {
        op.uncertain=true;setOutcome(it.absent);
      } else throw Error();
    } catch {
      if (operation.current === op) {op.uncertain=true;setOutcome(it.unknown);}
    } finally {
      if (operation.current === op || operation.current === null) setPending(false);
    }
  }
  return <details className="border rounded-lg p-3" data-testid="catalog-preparation">
    <summary className="font-semibold cursor-pointer">{t.title}</summary>
    <p>{ready || operation.current || outcome !== null ? it.mode : t.warning}</p>
    <button type="button" className="border rounded px-3 py-2 my-2" data-testid="catalog-load-source"
      disabled={sourceBusy || busy || pending || !!operation.current} onClick={() => void loadSource()}>
      {sourceBusy ? st.loading : st.load}
    </button>
    {sourceIssues !== null && <div role="status" data-testid="catalog-source-review">
      <p>{st.review}</p>
      {sourceIssues.length === 0 ? <p>{st.clean}</p> : <ul>
        {[["CATALOG_LOCATION_MAPPING_REQUIRED", st.locations], ["CATALOG_SUPPLIER_MAPPING_REQUIRED", st.suppliers],
          ["CATALOG_IMAGE_REVIEW_REQUIRED", st.photos]].map(([code, label]) => {
          const count = sourceIssues.filter(issue => issue === code).length;
          return count > 0 ? <li key={code}>{label}: {count}</li> : null;
        })}
        {sourceIssues.some(issue => !["CATALOG_LOCATION_MAPPING_REQUIRED", "CATALOG_SUPPLIER_MAPPING_REQUIRED",
          "CATALOG_IMAGE_REVIEW_REQUIRED"].includes(issue)) && <li>{st.other}</li>}
      </ul>}
    </div>}
    <label className="block">{t.input}<textarea data-testid="catalog-preparation-input"
      className="block w-full min-h-28 rounded border p-2 bg-background text-foreground"
      value={input} onChange={e => { epoch.current++; setInput(e.target.value); setResult(null); setError(null);setReady(false);setConfirmed(false);setSourceIssues(null); }} /></label>
    <button type="button" className="border rounded px-3 py-2" disabled={busy || sourceBusy || !input.trim()}
      onClick={() => void preview()}>{busy ? t.pending : t.preview}</button>
    {error && <p role="alert">{error}</p>}
    {result !== null && <p role="status">{t.result}: {result} · {ready || operation.current || outcome !== null ? it.scope : t.unverified}</p>}
    {ready && <div className="space-y-2 mt-2">
      <p>{it.photos}</p>
      <label className="flex gap-2 items-center"><input type="checkbox" data-testid="catalog-import-confirm"
        checked={confirmed} onChange={e=>setConfirmed(e.target.checked)} />{it.confirm}</label>
      <button type="button" className="border rounded px-3 py-2" data-testid="catalog-import"
        disabled={!confirmed || pending || !!operation.current} onClick={()=>void importCatalog()}>{it.save}</button>
    </div>}
    {operation.current && <div className="space-x-2 mt-2">
      <p>{it.locked}</p>
      <code data-testid="catalog-operation-id">{operation.current.id}</code>
      <button type="button" data-testid="catalog-result" disabled={pending} onClick={()=>void importCatalog(true)}>{it.check}</button>
      <button type="button" data-testid="catalog-retry" disabled={pending} onClick={()=>void importCatalog()}>{it.retry}</button>
    </div>}
    {outcome && <p role="status" data-testid="catalog-outcome">{outcome}</p>}
  </details>;
}
