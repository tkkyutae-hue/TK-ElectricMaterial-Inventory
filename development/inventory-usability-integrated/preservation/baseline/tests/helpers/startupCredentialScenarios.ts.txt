import test from "node:test";
import assert from "node:assert/strict";
import type * as Startup from "../../server/branches/startupPlan";

export function startupCredentialScenarios(api: typeof Startup) {
  const { readStartupPlan, verifyStartupPlan, endpointFingerprint } = api;
  const roles = ["tk_common_auth_runtime", "tk_savannah_business_runtime", "tk_indiana_business_runtime"];
  const keys = ["COMMON_AUTH_DATABASE_URL", "SAVANNAH_DATABASE_URL", "INDIANA_DATABASE_URL"];
  function fixture() {
    const urls = roles.map((role, i) => `postgresql://${role}:synthetic-only@target-${i}.invalid/synthetic?sslmode=require&channel_binding=require`);
    const approval: Startup.SharedApproval = {
      version: 1, evidenceId: "synthetic-review", masterAccountId: "synthetic-master",
      schemaAndRolesReviewed: true, recoveryReviewed: true, allowSharedStartup: true,
      identity: { targetId: "synthetic-common", endpointSha256: endpointFingerprint(urls[0]) },
      savannah: { targetId: "synthetic-savannah", endpointSha256: endpointFingerprint(urls[1]) },
      indiana: { targetId: "synthetic-indiana", endpointSha256: endpointFingerprint(urls[2]) },
    };
    const values: Record<string, string | undefined> = {
      TK_SERVER_MODE: "shared", TK_SHARED_APPROVAL_JSON: JSON.stringify(approval),
      ...Object.fromEntries(keys.map((key, i) => [key, urls[i]])),
      SESSION_SECRET: "synthetic-only-session-secret-32-characters", NODE_ENV: "production",
    };
    return { values, urls, approval };
  }
  function read(values: Record<string, string | undefined>) {
    return readStartupPlan(key => {
      assert.notEqual(key, "DATABASE_URL", "shared startup must never read the legacy owner connection");
      return values[key];
    });
  }
  function rejected(values: Record<string, string | undefined>) {
    assert.throws(() => read(values), (error: any) =>
      error.code === "SHARED_APPROVAL_REQUIRED" && error.status === 503 &&
      error.message === "SHARED_APPROVAL_REQUIRED");
  }
  test("offline guard strips real credentials and denies network/listeners", async () => {
    assert.deepEqual(Object.keys(process.env), ["STARTUP_PLAN_OFFLINE_GUARD"]);
    const net = await import("node:net");
    assert.throws(() => net.connect(5432), /STARTUP_PLAN_OFFLINE_ONLY/);
    assert.throws(() => net.createServer().listen(5000), /STARTUP_PLAN_OFFLINE_ONLY/);
  });
  test("default and explicit legacy read only the mode, not any credential", () => {
    for (const mode of [undefined, "legacy"]) {
      const reads: string[] = [];
      assert.deepEqual(readStartupPlan(key => { reads.push(key); return mode; }), { mode: "legacy" });
      assert.deepEqual(reads, ["TK_SERVER_MODE"]);
    }
    for (const mode of ["", "SHARED", "other"]) rejected({ TK_SERVER_MODE: mode });
  });
  test("shared pins three exact connections without reading DATABASE_URL", () => {
    const { values, urls } = fixture();
    const plan = read(values);
    assert.equal(plan.mode, "shared");
    if (plan.mode !== "shared") throw Error("unexpected legacy");
    assert.deepEqual([plan.identityConnection, plan.savannahConnection, plan.indianaConnection], urls);
    assert.equal(plan.secure, true);
    values.NODE_ENV = "development";
    assert.equal((read(values) as any).secure, false);
  });
  test("each missing/empty connection rejects even when legacy owner URL exists", () => {
    for (const key of keys) for (const value of [undefined, "", " "]) {
      const { values } = fixture();
      values.DATABASE_URL = "postgres://owner:synthetic-only@owner.invalid/synthetic";
      values[key] = value;
      rejected(values);
    }
  });
  test("owner, swapped and malformed principals cannot pass endpoint-only evidence", () => {
    for (const [i, key] of keys.entries()) for (const user of ["owner", "postgres", "neondb_owner", "", roles[(i + 1) % 3]]) {
      const { values } = fixture();
      values[key] = values[key]!.replace(roles[i], user);
      rejected(values);
    }
    for (const key of keys) for (const password of ["", "%20", "%ZZ"]) {
      const { values } = fixture();
      values[key] = values[key]!.replace("synthetic-only", password);
      rejected(values);
    }
  });
  test("URL overrides, options, duplicate TLS flags and insecure flags are refused", () => {
    for (const key of keys) for (const extra of [
      "user=owner", "password=override", "host=other.invalid", "port=5433", "database=other", "dbname=other",
      "options=-c%20role%3Dpostgres", "service=owner", "sslmode=disable", "sslmode=require",
      "channel_binding=prefer", "ssl=true", "USER=owner", "%75ser=owner",
    ]) {
      const { values } = fixture();
      values[key] += `&${extra}`;
      rejected(values);
    }
  });
  test("wrong targets, schemes, fragments and malformed URLs fail closed", () => {
    for (const key of keys) for (const transform of [
      (s: string) => s.replace(".invalid", "-wrong.invalid"),
      (s: string) => s.replace("/synthetic", "/other"),
      (s: string) => s.replace("postgresql:", "https:"),
      (s: string) => s + "#ignored",
      () => "not-a-url",
    ]) {
      const { values } = fixture();
      values[key] = transform(values[key]!);
      rejected(values);
    }
  });
  test("missing/malformed approvals, review gates and session secrets reject", () => {
    for (const input of [undefined, "", "{", "null", "{}", "[]"]) {
      const { values } = fixture(); values.TK_SHARED_APPROVAL_JSON = input; rejected(values);
    }
    for (const key of ["version", "schemaAndRolesReviewed", "recoveryReviewed", "allowSharedStartup", "masterAccountId", "evidenceId"]) {
      const { values, approval } = fixture();
      delete (approval as any)[key];
      values.TK_SHARED_APPROVAL_JSON = JSON.stringify(approval); rejected(values);
    }
    for (const secret of [undefined, "", "short"]) {
      const { values } = fixture(); values.SESSION_SECRET = secret; rejected(values);
    }
  });
  test("co-located common and Savannah keep different restricted principals", () => {
    const { values, approval } = fixture();
    values.COMMON_AUTH_DATABASE_URL = values.SAVANNAH_DATABASE_URL!.replace(roles[1], roles[0]);
    approval.identity = { ...approval.savannah };
    values.TK_SHARED_APPROVAL_JSON = JSON.stringify(approval);
    assert.equal(read(values).mode, "shared");
    values.COMMON_AUTH_DATABASE_URL = values.SAVANNAH_DATABASE_URL;
    rejected(values);
  });
  test("Indiana cannot share a physical endpoint or target ID with either source", () => {
    for (const source of ["identity", "savannah"] as const) {
      const { values, approval } = fixture();
      values.INDIANA_DATABASE_URL = values[source === "identity" ? keys[0] : keys[1]]!.replace(
        source === "identity" ? roles[0] : roles[1], roles[2]);
      approval.indiana = { ...approval[source] };
      values.TK_SHARED_APPROVAL_JSON = JSON.stringify(approval);
      rejected(values);
    }
    const { values, approval } = fixture();
    approval.indiana.targetId = approval.savannah.targetId;
    values.TK_SHARED_APPROVAL_JSON = JSON.stringify(approval); rejected(values);
  });
  test("inconsistent co-location evidence fails, while percent-encoded valid credentials work", () => {
    const { values, approval } = fixture();
    values.COMMON_AUTH_DATABASE_URL = values.SAVANNAH_DATABASE_URL!.replace(roles[1], roles[0]);
    approval.identity.endpointSha256 = approval.savannah.endpointSha256;
    values.TK_SHARED_APPROVAL_JSON = JSON.stringify(approval); rejected(values);
    const valid = fixture();
    valid.values.COMMON_AUTH_DATABASE_URL = valid.urls[0].replace("tk_common", "%74k_common").replace("synthetic-only", "synthetic%40only");
    assert.equal(read(valid.values).mode, "shared");
  });
  test("direct verifier also rejects owner credentials without leaking URL or password", () => {
    const plan = read(fixture().values) as Exclude<Startup.StartupPlan, { mode: "legacy" }>;
    plan.identityConnection = plan.identityConnection.replace(roles[0], "owner");
    assert.throws(() => verifyStartupPlan(plan), (error: any) =>
      error.code === "SHARED_APPROVAL_REQUIRED" && !error.message.includes("synthetic-only") && !error.message.includes(".invalid"));
  });
}
